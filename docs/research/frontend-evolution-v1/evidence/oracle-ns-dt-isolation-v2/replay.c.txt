#define F 629u
#define N (F*160u)
static int rd(const char*p,void*b,size_t n){FILE*f=fopen(p,"rb");if(!f)return 0;int ok=fread(b,1,n,f)==n&&fgetc(f)==EOF;return fclose(f)==0&&ok;}
static int wr(const char*p,const void*b,size_t n){FILE*f=fopen(p,"wb");if(!f)return 0;int ok=fwrite(b,1,n,f)==n;return fclose(f)==0&&ok;}
int main(int argc,char**argv){
 if(argc!=9)return 2;
 float *in=malloc(N*sizeof(float)),*echo=malloc(N*sizeof(float)),*out=malloc(N*sizeof(float)),*fields=malloc(F*4*sizeof(float));uint8_t far[F],dt[F];
 if(!in||!echo||!out||!fields)return 3;
 if(!rd(argv[1],in,N*sizeof(float))||!rd(argv[2],echo,N*sizeof(float))||!rd(argv[3],far,F)||!rd(argv[4],dt,F))return 4;
 for(unsigned i=0;i<N;i++)if(!isfinite(in[i])||!isfinite(echo[i]))return 5;
 for(unsigned i=0;i<F;i++)if(far[i]>1||dt[i]>1)return 5;
 float floor;if(!strcmp(argv[5],"call"))floor=0.12f;else if(!strcmp(argv[5],"assistant"))floor=0.18f;else return 6;
 struct timespec a,b;if(clock_gettime(CLOCK_THREAD_CPUTIME_ID,&a))return 7;
 ap_ns_state_t s;ap_ns_init(&s,160);
 for(unsigned f=0;f<F;f++){
  ap_ns_result_t r;ap_ns_process(&s,AP_ENHANCE_FULL,floor,in+f*160,echo+f*160,out+f*160,160,1,far[f],dt[f],&r);
  fields[f*4]=(float)r.frequency_res_active;fields[f*4+1]=r.residual_echo_gain;fields[f*4+2]=r.noise_rms_dbfs;fields[f*4+3]=r.speech_probability;
 }
 if(clock_gettime(CLOCK_THREAD_CPUTIME_ID,&b))return 7;
 double cpu=(double)(b.tv_sec-a.tv_sec)+1e-9*(double)(b.tv_nsec-a.tv_nsec);
 if(!wr(argv[6],out,N*sizeof(float))||!wr(argv[7],fields,F*4*sizeof(float)))return 8;
 FILE*f=fopen(argv[8],"w");if(!f)return 9;
 fprintf(f,"{\"thread_cpu_seconds\":%.17g,\"frames\":629,\"module_state_bytes\":%zu,\"single_observation\":true}\n",cpu,sizeof(s));if(fclose(f))return 9;
 free(in);free(echo);free(out);free(fields);return 0;
}
